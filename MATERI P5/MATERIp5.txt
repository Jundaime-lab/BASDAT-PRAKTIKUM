);

CREATE TABLE binatang_2 (
	id_binatang2 INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
	binatang2 VARCHAR(100)
);

-- INNER JOIN
SELECT
	binatang_1.id_binatang1,
	binatang_1.binatang1,
	
	binatang_2.id_binatang2,
	binatang_2.binatang2
FROM
	binatang_1
INNER JOIN binatang_2
ON binatang_1.binatang1 = binatang_2.binatang2;

-- LEFT JOIN
SELECT
	binatang_1.id_binatang1,
	binatang_1.binatang1,
	
	binatang_2.id_binatang2,
	binatang_2.binatang2
FROM
	binatang_1
LEFT JOIN binatang_2
ON binatang_1.binatang1 = binatang_2.binatang2;

-- RIGHT JOIN
SELECT
	binatang_1.id_binatang1,
	binatang_1.binatang1,
	
	binatang_2.id_binatang2,
	binatang_2.binatang2
FROM
	binatang_1
RIGHT JOIN binatang_2
ON binatang_1.binatang1 = binatang_2.binatang2;

-- FULL JOIN
SELECT
	binatang_1.id_binatang1,
	binatang_1.binatang1,
	
	binatang_2.id_binatang2,
	binatang_2.binatang2
FROM
	binatang_1
FULL JOIN binatang_2
ON binatang_1.binatang1 = binatang_2.binatang2;

-- CROSS JOIN
SELECT * FROM binatang_1 CROSS JOIN binatang_2;
